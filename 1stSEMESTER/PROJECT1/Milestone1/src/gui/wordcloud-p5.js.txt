let clouds = [];
let cloudAtual = 0;
let ultimaTroca = 0;
let fonte;
 
function setup() {
  createCanvas(CANVAS_W, CANVAS_H);
  textFont("Georgia, serif");
  angleMode(DEGREES);
 
  // Cria 6 clouds temáticas diferentes.
  // Cada uma usa palavras diferentes, mas a lista completa é aproveitada ao longo da rotação.
  const temas = [
    { nome: "Educação e Escolas", filtro: /school|teacher|student|education|class|learn|curricul|principal|classroom|grade|attendance|homework|lesson/i },
    { nome: "Avaliação e Dados", filtro: /assessment|test|score|data|variable|response|questionnaire|survey|sampling|weight|scaling|SPSS|validity|reliability|item/i },
    { nome: "Matemática e Leitura", filtro: /math|reading|literacy|numeracy|science|problem|reasoning|critical|book|library|text|language/i },
    { nome: "Família e Contexto", filtro: /parent|family|home|household|income|wealth|poverty|sibling|immigrant|migration|birth|gender|age|language/i },
    { nome: "Bem-estar e Motivação", filtro: /motivation|interest|anxiety|wellbeing|belonging|friendship|bullying|resilience|perseverance|confidence|satisfaction|engagement|support/i },
    { nome: "Tecnologia e Futuro", filtro: /ICT|digital|computer|internet|technology|software|online|global|sustainab|environment|career|future|employment|work|finance/i }
  ];
 
  temas.forEach((tema, i) => {
    let base = PALAVRAS.filter(p => tema.filtro.test(p));
 
    // Garante variedade: se o tema tiver poucas correspondências, mistura com o resto.
    if (base.length < 45) {
      const outras = PALAVRAS.filter(p => !base.includes(p));
      base = base.concat(shuffle(outras).slice(0, 45 - base.length));
    }
 
    // Repete palavras para encher bem o canvas.
    let lista = [];
    while (lista.length < 110) {
      lista = lista.concat(shuffle(base.slice()));
    }
    lista = lista.slice(0, 110);
 
    clouds.push(criarCloud(lista, tema.nome, i));
  });
 
  ultimaTroca = millis();
}
 
function draw() {
  background(18, 22, 34);
 
  // Troca de cloud a cada 5 segundos
  if (millis() - ultimaTroca >= DURACAO) {
    cloudAtual = (cloudAtual + 1) % clouds.length;
    ultimaTroca = millis();
  }
 
  const cloud = clouds[cloudAtual];
 
  // Transição suave de opacidade
  const decorrido = millis() - ultimaTroca;
  const alpha = decorrido < 500 ? map(decorrido, 0, 500, 0, 255) : 255;
 
  desenharCloud(cloud, alpha);
  desenharTitulo(cloud.titulo, alpha);
  desenharIndicador();
}
 
function criarCloud(palavras, titulo, seed) {
  randomSeed(seed * 1000 + 7);
  const colocadas = [];
 
  for (const palavra of palavras) {
    const tentativa = tentarColocarPalavra(palavra, colocadas);
    if (tentativa) colocadas.push(tentativa);
  }
 
  return { titulo, palavras: colocadas };
}
 
function tentarColocarPalavra(palavra, colocadas) {
  const tamanhoBase = map(palavra.length, 3, 24, 30, 13);
  const tamanho = random(tamanhoBase * 0.75, tamanhoBase * 1.25);
 
  for (let i = 0; i < 220; i++) {
    const x = random(70, CANVAS_W - 70);
    const y = random(95, CANVAS_H - 55);
    const rotacao = random() < 0.78 ? 0 : random([-90, 90]);
    const cor = corAleatoria();
 
    textSize(tamanho);
    const w = textWidth(palavra);
    const h = tamanho;
 
    // Delimitação simples para evitar sobreposições
    const caixa = {
      x: x - w / 2 - 4,
      y: y - h / 2 - 4,
      w: w + 8,
      h: h + 8,
      rotacao
    };
 
    if (!colide(caixa, colocadas)) {
      return { texto: palavra, x, y, tamanho, rotacao, cor, caixa };
    }
  }
 
  return null;
}
 
function colide(a, lista) {
  for (const b of lista) {
    if (
      a.x < b.caixa.x + b.caixa.w &&
      a.x + a.w > b.caixa.x &&
      a.y < b.caixa.y + b.caixa.h &&
      a.y + a.h > b.caixa.y
    ) {
      return true;
    }
  }
  return false;
}
 
function desenharCloud(cloud, alpha) {
  push();
  for (const p of cloud.palavras) {
    push();
    translate(p.x, p.y);
    rotate(p.rotacao);
    noStroke();
    fill(red(p.cor), green(p.cor), blue(p.cor), alpha);
    textSize(p.tamanho);
    textAlign(CENTER, CENTER);
    text(p.texto, 0, 0);
    pop();
  }
  pop();
}
 
function desenharTitulo(titulo, alpha) {
  push();
  noStroke();
  fill(235, 240, 255, alpha);
  textSize(24);
  textAlign(CENTER, TOP);
  text(titulo, CANVAS_W / 2, 26);
 
  stroke(90, 110, 150, alpha * 0.6);
  strokeWeight(1);
  line(140, 66, CANVAS_W - 140, 66);
  pop();
}
 
function desenharIndicador() {
  const total = clouds.length;
  const espacamento = 18;
  const inicioX = CANVAS_W / 2 - ((total - 1) * espacamento) / 2;
 
  push();
  noStroke();
 
  for (let i = 0; i < total; i++) {
    if (i === cloudAtual) {
      fill(120, 190, 255);
    } else {
      fill(70, 85, 110);
    }
 
    circle(inicioX + i * espacamento, CANVAS_H - 22, 8);
  }
 
  pop();
}
 
function corAleatoria() {
  const paleta = [
    color(126, 217, 255),
    color(255, 209, 102),
    color(255, 138, 138),
    color(160, 255, 189),
    color(206, 178, 255),
    color(255, 255, 255),
    color(255, 177, 224)
  ];
  return random(paleta);
}
 
function shuffle(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = floor(random(i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
